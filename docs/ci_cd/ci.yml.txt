# CI gate for SSMS — runs on every push and pull request.
#
# Purpose: nothing broken reaches `main`. Every job here is a BLOCKER;
# if one fails the pull request cannot be merged.
#
# Motivating incident: a read-only-map crash (P41) reached a real phone
# and cost four debugging rounds. `flutter test` now runs the regression
# test that reproduces it, on every single push.
name: CI

on:
  push:
    branches: [main]
  pull_request:
  workflow_dispatch: # lets you run it by hand from the Actions tab

# A newer push to the same branch cancels the older run — saves minutes
# and means the result you see always matches the latest commit.
concurrency:
  group: ci-${{ github.ref }}
  cancel-in-progress: true

env:
  FLUTTER_VERSION: '3.27.0' # pubspec requires >=3.27.0
  APP_DIR: Mobile/wbws_flutter_app

jobs:
  # ── 1. Static analysis + unit tests ───────────────────────────────
  flutter-test:
    name: Analyze and test (Flutter)
    runs-on: ubuntu-latest
    timeout-minutes: 20
    steps:
      - uses: actions/checkout@v4

      - uses: subosito/flutter-action@v2
        with:
          flutter-version: ${{ env.FLUTTER_VERSION }}
          channel: stable
          cache: true # caches the SDK between runs

      - name: Install dependencies
        working-directory: ${{ env.APP_DIR }}
        run: flutter pub get

      # Formatting is checked but NOT allowed to fail the build: style is
      # not correctness, and a failing gate people learn to ignore is
      # worse than no gate. Revisit once the team agrees on a style.
      - name: Check formatting (informational)
        working-directory: ${{ env.APP_DIR }}
        run: dart format --output=none --set-exit-if-changed . || true

      - name: Analyze
        working-directory: ${{ env.APP_DIR }}
        run: flutter analyze

      # P49: catches unguarded access on nullable, non-final fields.
      # `flutter analyze` already covers this in CI, but the linter also
      # runs in environments that cannot host the Flutter SDK (the
      # development sandbox), which is where three build breaks
      # originated. Cheap, precise, and no false positives.
      - name: Nullable-field lint
        working-directory: ${{ env.APP_DIR }}/tool
        run: |
          dart pub get
          dart run nullable_field_lint.dart ../lib

      - name: Run tests
        working-directory: ${{ env.APP_DIR }}
        run: flutter test --reporter expanded

  # ── 2. Real release build ─────────────────────────────────────────
  # Separate from tests on purpose: `flutter analyze` does NOT catch
  # everything the compiler does. A build that fails here means the APK
  # could never have been produced.
  flutter-build:
    name: Build release APK
    runs-on: ubuntu-latest
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - uses: subosito/flutter-action@v2
        with:
          flutter-version: ${{ env.FLUTTER_VERSION }}
          channel: stable
          cache: true

      - name: Install dependencies
        working-directory: ${{ env.APP_DIR }}
        run: flutter pub get

      # Unsigned: this proves the code COMPILES. Signing needs the
      # keystore secret and belongs to the release workflow (Phase 2).
      - name: Build APK (unsigned, debug-signed)
        working-directory: ${{ env.APP_DIR }}
        run: flutter build apk --release

      # Keeps the APK for 14 days so you can install a PR build on a real
      # phone before merging it.
      - name: Upload APK artifact
        uses: actions/upload-artifact@v4
        with:
          name: app-release-apk
          path: ${{ env.APP_DIR }}/build/app/outputs/flutter-apk/app-release.apk
          retention-days: 14
          if-no-files-found: error

  # ── 3. PHP syntax check ───────────────────────────────────────────
  # The backend is deployed by `git pull`, so a syntax error goes live
  # and takes the site down with a blank page. `php -l` makes that
  # impossible to merge.
  php-lint:
    name: Lint PHP
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - uses: actions/checkout@v4

      - uses: shivammathur/setup-php@v2
        with:
          php-version: '8.1'
          coverage: none

      - name: Syntax-check every PHP file
        run: |
          set -euo pipefail
          fail=0
          # vendor/ is third-party and already released; linting it adds
          # minutes and can fail on files needing a newer PHP.
          while IFS= read -r -d '' f; do
            if ! php -l "$f" > /dev/null 2>&1; then
              echo "::error file=$f::PHP syntax error"
              php -l "$f" 2>&1 | sed 's/^/    /'
              fail=1
            fi
          done < <(find . -name '*.php' -not -path './vendor/*' -print0)
          if [ "$fail" -ne 0 ]; then
            echo "PHP syntax errors found — see annotations above."
            exit 1
          fi
          echo "All PHP files parse cleanly."

  # ── 4. Secret / large-file guard ──────────────────────────────────
  # Credentials in Git are permanent — rewriting history does not remove
  # them from clones or forks. Cheapest possible check, highest payoff.
  hygiene:
    name: Repo hygiene
    runs-on: ubuntu-latest
    timeout-minutes: 5
    steps:
      - uses: actions/checkout@v4

      - name: Block committed secrets and APKs
        run: |
          set -euo pipefail
          bad=0
          # Files that must never be tracked, matching .gitignore intent.
          while IFS= read -r f; do
            case "$f" in
              *.apk|*.jks|*.keystore|.env|*_env.php|.fkss_env.php|.wbws_env.php)
                echo "::error file=$f::must never be committed"
                bad=1 ;;
            esac
          done < <(git ls-files)

          # Anything over 20 MB bloats every future clone, permanently.
          while IFS= read -r f; do
            [ -f "$f" ] || continue
            sz=$(stat -c%s "$f")
            if [ "$sz" -gt 20971520 ]; then
              echo "::error file=$f::file is $((sz/1048576)) MB — too large for Git"
              bad=1
            fi
          done < <(git ls-files)

          [ "$bad" -eq 0 ] || exit 1
          echo "No secrets or oversized files tracked."
