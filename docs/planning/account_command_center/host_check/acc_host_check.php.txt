<?php
/**
 * ACC — READ-ONLY production hosting check (D3 prerequisite).
 *
 * WHAT IT DOES: prints facts about PHP, extensions, OpenSSL EC support,
 * password hashing, MariaDB version/privileges/settings. It runs ONLY
 * SELECT / SHOW statements. It creates nothing, writes nothing, changes nothing.
 *
 * HOW TO USE (once):
 *   1. Rename this file to something unguessable, e.g. acc_check_7f3k2q9z.php
 *   2. Edit ACC_CHECK_KEY below to a long random value.
 *   3. Upload it NEXT TO config.php (site root, same folder as index.php).
 *   4. Open https://<your-site>/acc_check_7f3k2q9z.php?key=<ACC_CHECK_KEY>
 *   5. Copy the whole output (it is JSON) and send it back.
 *   6. DELETE the file from the server.
 */
declare(strict_types=1);

const ACC_CHECK_KEY = 'CHANGE-ME-TO-A-LONG-RANDOM-STRING';

if (!isset($_GET['key']) || !hash_equals(ACC_CHECK_KEY, (string)$_GET['key']) || strlen(ACC_CHECK_KEY) < 20 || str_starts_with(ACC_CHECK_KEY, 'CHANGE-ME')) {
    http_response_code(404);
    exit('Not found');
}

header('Content-Type: application/json; charset=utf-8');
header('Cache-Control: no-store');

$out = ['generated_at_utc' => gmdate('c'), 'checks' => []];
$ok  = static function (string $name, $value, string $note = '') use (&$out): void {
    $out['checks'][$name] = ['value' => $value, 'note' => $note];
};

/* ---------- PHP runtime ---------- */
$ok('php.version', PHP_VERSION, 'league/oauth2-server v9 needs >= 8.2 (v8.5 works on 8.1)');
$ok('php.sapi', PHP_SAPI);
$ok('php.memory_limit', ini_get('memory_limit'));
$ok('php.max_execution_time', ini_get('max_execution_time'));
$ok('php.upload_max_filesize', ini_get('upload_max_filesize'));
$ok('php.post_max_size', ini_get('post_max_size'));
$ok('php.disable_functions', ini_get('disable_functions'));
$ok('php.open_basedir', ini_get('open_basedir'));
$ok('server.https', !empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off');
$ok('server.software', $_SERVER['SERVER_SOFTWARE'] ?? null);

/* ---------- Extensions ---------- */
foreach (['openssl','json','mbstring','sodium','gmp','bcmath','mysqli','pdo_mysql','zlib','curl','intl','fileinfo','gd'] as $ext) {
    $ok("ext.$ext", extension_loaded($ext));
}
$ok('openssl.version_text', defined('OPENSSL_VERSION_TEXT') ? OPENSSL_VERSION_TEXT : null);

/* ---------- OpenSSL EC P-256: generate (in memory), sign, verify ---------- */
$ecResult = ['generate' => false, 'sign' => false, 'verify' => false, 'error' => null];
try {
    if (function_exists('openssl_pkey_new')) {
        $key = @openssl_pkey_new(['curve_name' => 'prime256v1', 'private_key_type' => OPENSSL_KEYTYPE_EC]);
        if ($key !== false) {
            $ecResult['generate'] = true;
            $details = openssl_pkey_get_details($key);
            $pub = $details['key'] ?? null;
            $msg = 'acc-host-check';
            $sig = '';
            if (@openssl_sign($msg, $sig, $key, OPENSSL_ALGO_SHA256)) {
                $ecResult['sign'] = true;
                $ecResult['verify'] = (openssl_verify($msg, $sig, $pub, OPENSSL_ALGO_SHA256) === 1);
            }
        } else {
            $ecResult['error'] = openssl_error_string() ?: 'openssl_pkey_new failed';
        }
    }
} catch (Throwable $e) { $ecResult['error'] = get_class($e); }
$ok('openssl.ec_p256', $ecResult, 'All three must be true for device-proof verification (ES256)');

/* ---------- Password hashing ---------- */
$ok('password.argon2id_available', defined('PASSWORD_ARGON2ID'), 'Preferred for PIN hashing; bcrypt fallback otherwise');
$ok('password.bcrypt_available', defined('PASSWORD_BCRYPT'));
try { $ok('random_bytes.ok', strlen(random_bytes(16)) === 16); } catch (Throwable $e) { $ok('random_bytes.ok', false, get_class($e)); }
$ok('hash.algos_has_sha256', in_array('sha256', hash_algos(), true));

/* ---------- Database (READ-ONLY) ---------- */
$db = ['connected' => false];
try {
    // Load the application's own config to obtain DB constants (no side effects beyond what every page does).
    if (!defined('DB_HOST')) {
        $cfg = __DIR__ . '/config.php';
        if (is_file($cfg)) { ob_start(); require_once $cfg; ob_end_clean(); }
    }
    if (defined('DB_HOST') && defined('DB_USER') && defined('DB_PASS') && defined('DB_NAME')) {
        mysqli_report(MYSQLI_REPORT_ERROR | MYSQLI_REPORT_STRICT);
        $m = new mysqli(DB_HOST, DB_USER, DB_PASS, DB_NAME);
        $db['connected'] = true;
        $one = static function (string $sql) use ($m) {
            $r = $m->query($sql); $row = $r ? $r->fetch_row() : null; if ($r) $r->free(); return $row[0] ?? null;
        };
        $db['server_version']        = $one('SELECT VERSION()');
        $db['current_user']          = $one('SELECT CURRENT_USER()');
        $db['sql_mode']              = $one('SELECT @@sql_mode');
        $db['default_storage_engine']= $one('SELECT @@default_storage_engine');
        $db['character_set_server']  = $one('SELECT @@character_set_server');
        $db['collation_server']      = $one('SELECT @@collation_server');
        $db['max_allowed_packet']    = $one('SELECT @@max_allowed_packet');
        $db['innodb_file_per_table'] = $one('SELECT @@innodb_file_per_table');
        $db['event_scheduler']       = $one('SELECT @@event_scheduler');
        $db['log_bin']               = $one('SELECT @@log_bin');
        $db['log_bin_trust_function_creators'] = $one('SELECT @@log_bin_trust_function_creators');
        $db['time_zone']             = $one('SELECT @@time_zone');
        $db['utc_now']               = $one('SELECT UTC_TIMESTAMP()');
        // Privileges (SHOW GRANTS is read-only)
        $grants = [];
        $r = $m->query('SHOW GRANTS FOR CURRENT_USER()');
        while ($r && ($row = $r->fetch_row())) { $grants[] = $row[0]; }
        if ($r) $r->free();
        $db['grants'] = $grants;
        $gtxt = implode("\n", $grants);
        $db['priv.trigger']          = (bool)preg_match('/\bALL PRIVILEGES\b|\bTRIGGER\b/i', $gtxt);
        $db['priv.create_routine']   = (bool)preg_match('/\bALL PRIVILEGES\b|\bCREATE ROUTINE\b/i', $gtxt);
        $db['priv.event']            = (bool)preg_match('/\bALL PRIVILEGES\b|\bEVENT\b/i', $gtxt);
        $db['priv.delete']           = (bool)preg_match('/\bALL PRIVILEGES\b|\bDELETE\b/i', $gtxt);
        // Existing triggers in this schema (read from information_schema)
        $db['existing_triggers']     = (int)$one("SELECT COUNT(*) FROM information_schema.TRIGGERS WHERE TRIGGER_SCHEMA = DATABASE()");
        $db['existing_routines']     = (int)$one("SELECT COUNT(*) FROM information_schema.ROUTINES WHERE ROUTINE_SCHEMA = DATABASE()");
        $db['fulltext_supported_innodb'] = true; // InnoDB FULLTEXT exists since 5.6/10.0; version string above confirms
        // Schema facts ACC depends on (presence only)
        $tables = [];
        foreach (['users','members','guardians','member_guardians','activity_logs','api_refresh_sessions','api_idempotency'] as $t) {
            $tables[$t] = ((int)$one("SELECT COUNT(*) FROM information_schema.TABLES WHERE TABLE_SCHEMA = DATABASE() AND TABLE_NAME = '" . $m->real_escape_string($t) . "'")) === 1;
        }
        $db['required_tables_present'] = $tables;
        $db['schema_size_mb'] = $one("SELECT ROUND(SUM(DATA_LENGTH + INDEX_LENGTH)/1024/1024, 1) FROM information_schema.TABLES WHERE TABLE_SCHEMA = DATABASE()");
        $m->close();
    } else {
        $db['error'] = 'DB constants not available (config.php/env not found next to this script)';
    }
} catch (Throwable $e) {
    $db['error'] = get_class($e); // message withheld on purpose (no disclosure)
}
$ok('db', $db, 'priv.trigger=true means append-only triggers are possible; otherwise hash-chain + verification only');

/* ---------- Filesystem ---------- */
$ok('fs.disk_free_mb', is_callable('disk_free_space') ? round((float)@disk_free_space(__DIR__) / 1048576) : null);
$ok('fs.vendor_present', is_dir(__DIR__ . '/vendor'));
$ok('fs.env_file_outside_webroot', is_file(dirname(__DIR__) . '/.fkss_env.php') || is_file(dirname(__DIR__) . '/.wbws_env.php'));

/* ---------- Summary ---------- */
$c = $out['checks'];
$out['summary'] = [
    'php_ok_for_league_v9'   => version_compare(PHP_VERSION, '8.2.0', '>='),
    'php_ok_for_league_v8'   => version_compare(PHP_VERSION, '8.1.0', '>='),
    'ec_p256_ok'             => ($c['openssl.ec_p256']['value']['verify'] ?? false) === true,
    'argon2id'               => $c['password.argon2id_available']['value'],
    'triggers_possible'      => $c['db']['value']['priv.trigger'] ?? null,
    'db_connected'           => $c['db']['value']['connected'] ?? false,
    'mariadb_version'        => $c['db']['value']['server_version'] ?? null,
];
echo json_encode($out, JSON_PRETTY_PRINT | JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE);
